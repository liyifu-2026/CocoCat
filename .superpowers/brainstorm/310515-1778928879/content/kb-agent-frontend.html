kb-agent 注册（Resident Agent）
# 注册为 Resident Agent，与 Coco 平级
class AgentRole(Enum):
    RESIDENT = "resident"
    WORKER = "worker"

# config/residents/kb-agent.yaml
id: kb-agent
name: 知识库管理员
role: resident
page: /knowledge # 绑定页面
skills: [knowledge-ingestion]
tools: [search_kb, read_wiki, write_wiki,
        run_dedup, run_lint, gen_overview,
        cascade_del, get_graph, call_worker]
cron: [kb:lint@24h, kb:dedup@7d, kb:overview@7d]
